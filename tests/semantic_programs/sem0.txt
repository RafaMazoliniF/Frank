
programa teste_recursao_fatorial;
var n_global, resultado: inteiro;

funcao fatorial: inteiro;
var n_local: inteiro; 
inicio
    n_local := n_global;
    
    se n_local <= 1 entao
        fatorial := 1 
    senao
    inicio
        n_global := n_local - 1; 
        fatorial := n_local * fatorial;
    fim;
fim;

inicio
    leia(n_global); 
    
    resultado := fatorial;
    
    escreva(resultado);
fim.
